cmake_minimum_required(VERSION 3.23)
project(ZoneEngineProbe LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set_source_files_properties(Host/Stack.c PROPERTIES COMPILE_DEFINITIONS _GNU_SOURCE=1)
set(WABT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../Tools/wabt" CACHE PATH "Pinned patched WABT checkout")
set(ZONE_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Generated" CACHE PATH "wasm2c output directory")
find_package(Threads REQUIRED)
file(GLOB ENGINE_C CONFIGURE_DEPENDS "${ZONE_GENERATED_DIR}/xray_*.c")
if(NOT ENGINE_C)
  message(FATAL_ERROR "Generate the engine first with Scripts/prepare_engine.py")
endif()
add_library(xray_aot STATIC ${ENGINE_C} Host/Atomics.c Host/Stack.c
  "${WABT_ROOT}/wasm2c/wasm-rt-impl.c"
  "${WABT_ROOT}/wasm2c/wasm-rt-mem-impl.c"
  "${WABT_ROOT}/wasm2c/wasm-rt-exceptions-impl.c")
target_include_directories(xray_aot PUBLIC "${ZONE_GENERATED_DIR}" "${WABT_ROOT}/wasm2c" "${WABT_ROOT}/third_party/simde" Host)
target_compile_definitions(xray_aot PUBLIC WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_SKIP_SIGNAL_RECOVERY=1 WASM_RT_MAX_CALL_STACK_DEPTH=4096)
set(ZONE_ENGINE_OPTIMIZATION "0" CACHE STRING "Engine C optimization level (0 for bring-up, 1 for faster startup)")
if(NOT ZONE_ENGINE_OPTIMIZATION MATCHES "^[0123]$")
  message(FATAL_ERROR "ZONE_ENGINE_OPTIMIZATION must be 0, 1, 2 or 3")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(xray_aot PRIVATE -O${ZONE_ENGINE_OPTIMIZATION} -Werror=incompatible-pointer-types -Werror=implicit-function-declaration
    -include "${CMAKE_CURRENT_SOURCE_DIR}/Host/NativeRuntime.h")
  # A few very large generated units take disproportionate optimization time
  # in GCC. Diagnostics can opt individual units out while retaining the
  # existing objects and optimization of the rest of the engine.
  set(ZONE_ENGINE_UNOPTIMIZED_UNITS "" CACHE STRING "Generated C basenames compiled at O0 for bring-up")
  foreach(unit IN LISTS ZONE_ENGINE_UNOPTIMIZED_UNITS)
    set(source "${ZONE_GENERATED_DIR}/${unit}")
    if(NOT source IN_LIST ENGINE_C)
      message(FATAL_ERROR "Unknown generated engine unit: ${unit}")
    endif()
    set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS -O0)
  endforeach()
endif()
target_link_libraries(xray_aot PUBLIC Threads::Threads m)
add_executable(zone_atomics_test ../Tests/atomics_test.c Host/Atomics.c
  "${WABT_ROOT}/wasm2c/wasm-rt-impl.c" "${WABT_ROOT}/wasm2c/wasm-rt-mem-impl.c")
target_include_directories(zone_atomics_test PRIVATE "${WABT_ROOT}/wasm2c" "${WABT_ROOT}/third_party/simde" Host)
target_compile_definitions(zone_atomics_test PRIVATE WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_SKIP_SIGNAL_RECOVERY=1)
target_link_libraries(zone_atomics_test PRIVATE Threads::Threads m)
target_compile_options(zone_atomics_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME wasm_atomics COMMAND zone_atomics_test)
set(EXCEPTION_FIXTURE "${CMAKE_CURRENT_BINARY_DIR}/exception-fixture")
add_custom_command(OUTPUT "${EXCEPTION_FIXTURE}/exception_depth.c" "${EXCEPTION_FIXTURE}/exception_depth.h"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${EXCEPTION_FIXTURE}"
  COMMAND "${WABT_ROOT}/build/wat2wasm" "${CMAKE_CURRENT_SOURCE_DIR}/../Tests/exception_depth.wat" -o "${EXCEPTION_FIXTURE}/exception_depth.wasm"
  COMMAND "${WABT_ROOT}/build/wasm2c" "${EXCEPTION_FIXTURE}/exception_depth.wasm" --module-name=exception_depth -o "${EXCEPTION_FIXTURE}/exception_depth.c"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../Tests/exception_depth.wat" "${WABT_ROOT}/build/wasm2c")
add_executable(zone_exception_test ../Tests/exception_depth.c "${EXCEPTION_FIXTURE}/exception_depth.c")
target_include_directories(zone_exception_test PRIVATE "${EXCEPTION_FIXTURE}")
target_compile_options(zone_exception_test PRIVATE -UNDEBUG)
target_link_libraries(zone_exception_test PRIVATE xray_aot)
add_test(NAME exception_depth COMMAND zone_exception_test)
if(EXISTS "${ZONE_GENERATED_DIR}/cpu_probe.c")
  add_executable(zone_cpu_probe "${ZONE_GENERATED_DIR}/cpu_probe.c")
  target_link_libraries(zone_cpu_probe PRIVATE xray_aot)
endif()
if(EXISTS "${ZONE_GENERATED_DIR}/native_imports.c")
  find_package(OpenAL REQUIRED)
  add_library(zone_native_host STATIC Host/Runtime.c Host/Threads.c Host/Files.c Host/Memory.c Host/Asm.c Host/Window.c
    Host/Audio.c Host/Network.c "${ZONE_GENERATED_DIR}/native_imports.c")
  target_include_directories(zone_native_host PRIVATE ${OPENAL_INCLUDE_DIR})
  target_link_libraries(zone_native_host PUBLIC xray_aot ${OPENAL_LIBRARY})
  target_compile_definitions(zone_native_host PUBLIC _DARWIN_C_SOURCE=1 PRIVATE _DEFAULT_SOURCE)
  target_compile_options(zone_native_host PRIVATE -Wall -Wextra -Werror=incompatible-pointer-types -Werror=implicit-function-declaration)
  option(ZONE_NATIVE_GLES "Enable the checked native GLES 3 host" OFF)
  if(ZONE_NATIVE_GLES)
    if(NOT EXISTS "${ZONE_GENERATED_DIR}/native_gles.c")
      message(FATAL_ERROR "Generate GLES bindings with Scripts/native_host.py --gles")
    endif()
    target_sources(zone_native_host PRIVATE Host/Graphics.c "${ZONE_GENERATED_DIR}/native_gles.c")
    target_compile_definitions(zone_native_host PUBLIC ZONE_NATIVE_GLES=1)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
      enable_language(OBJC)
      target_sources(zone_native_host PRIVATE Host/GraphicsIOS.m)
      set_source_files_properties(Host/GraphicsIOS.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
      find_library(ZONE_GLES_LIBRARY OpenGLES REQUIRED)
      find_library(ZONE_QUARTZCORE_LIBRARY QuartzCore REQUIRED)
      find_library(ZONE_FOUNDATION_LIBRARY Foundation REQUIRED)
      target_link_libraries(zone_native_host PUBLIC ${ZONE_GLES_LIBRARY} ${ZONE_QUARTZCORE_LIBRARY} ${ZONE_FOUNDATION_LIBRARY})
    elseif(NOT APPLE)
      find_path(ZONE_GLES_INCLUDE_DIR GLES3/gl3.h REQUIRED)
      find_library(ZONE_GLES_LIBRARY GLESv2 REQUIRED)
      find_library(ZONE_EGL_LIBRARY EGL REQUIRED)
      target_include_directories(zone_native_host PUBLIC ${ZONE_GLES_INCLUDE_DIR})
      target_sources(zone_native_host PRIVATE Host/GraphicsEGL.c)
      target_link_libraries(zone_native_host PUBLIC ${ZONE_GLES_LIBRARY} ${ZONE_EGL_LIBRARY})
      add_executable(zone_native_gles_test ../Tests/native_gles.c)
      target_compile_options(zone_native_gles_test PRIVATE -UNDEBUG)
      target_link_libraries(zone_native_gles_test PRIVATE zone_native_host)
      add_test(NAME native_gles COMMAND zone_native_gles_test)
    else()
      message(FATAL_ERROR "The GLES provider supports iOS and Linux, not macOS")
    endif()
  endif()
  add_executable(zone_native_boot Host/Boot.c)
  target_link_libraries(zone_native_boot PRIVATE zone_native_host)
  if(UNIX AND NOT APPLE)
    target_link_options(zone_native_boot PRIVATE -rdynamic)
  endif()
  add_executable(zone_native_host_test ../Tests/native_host.c)
  target_include_directories(zone_native_host_test PRIVATE ${OPENAL_INCLUDE_DIR})
  target_compile_options(zone_native_host_test PRIVATE -UNDEBUG)
  target_link_libraries(zone_native_host_test PRIVATE zone_native_host)
  add_test(NAME native_host COMMAND zone_native_host_test)
  set_tests_properties(native_host PROPERTIES ENVIRONMENT "ALSOFT_DRIVERS=null"
    PASS_REGULAR_EXPRESSION "Native filesystem:.*70,000 handle cycles passed")
endif()
